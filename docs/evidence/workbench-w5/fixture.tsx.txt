import React from 'react';
import { createRoot } from 'react-dom/client';
import { MemoryRouter, Route, Routes } from 'react-router';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { MotionProvider } from '@appliance.sh/ui/motion-provider';
import { HostProvider } from 'REPO/packages/app/src/providers/host-provider';
import { WorkbenchDetailPage } from 'REPO/packages/app/src/pages/workbench/detail';
import 'REPO/packages/app/src/styles.css';
const directory = '/fixture/workspace';
const listeners = new Set();
let snapshot = null;
const source = {getSnapshot: () => snapshot, subscribe: fn => {listeners.add(fn); return () => listeners.delete(fn);}};
const proposal = {source:'fixture', directory, proposalId:'ready-1', sessionId:'agent-fixture-1', sessionStatus:'done', summary:'Simulated captured result: changes prepared for human inspection. This does not certify tests or authorize packaging.'};
window.emitProposal = (status='done', scope=directory) => {snapshot={...proposal, directory:scope, sessionStatus:status}; listeners.forEach(fn=>fn());};
window.packageCalls=0;
const workspace={directory,binding:{version:1},sessions:[{sessionId:proposal.sessionId,status:'done'}]};
const host={platform:'macos',getConfig:async()=>({clusters:[]}),workbench:{open:async()=>workspace},workbenchPackage:{package:async()=>{window.packageCalls++;window.backendStart=performance.now();await new Promise(r=>setTimeout(r,180));window.backendEnd=performance.now();throw new Error('Fixture-only package failure; no backend was called.');}}};
const cache=new QueryClient({defaultOptions:{queries:{retry:false,staleTime:Infinity}}});
cache.setQueryData(['workbench',directory],workspace);
cache.setQueryData(['host','config'],{clusters:[]});
createRoot(document.getElementById('root')).render(<MotionProvider><HostProvider host={host}><QueryClientProvider client={cache}><MemoryRouter initialEntries={['/workbench/'+encodeURIComponent(directory)]}><p style={{padding:16}}>Fixture-only F3 bridge — simulated session, result and package host; no live authorization path.</p><Routes><Route path="/workbench/:workspaceId" element={<WorkbenchDetailPage readinessSource={source}/>}/></Routes></MemoryRouter></QueryClientProvider></HostProvider></MotionProvider>);
